\section*{Highlights}
\noindent\textbf{Porous polyimide/hexagonal boron nitride films by non-solvent induced phase separation: a three-phase Lewis--Nielsen description of dielectric and thermal properties}
\begin{itemize}
\item Porous PI/h-BN films made by NIPS over the full 0--70~wt\% h-BN range
\item Three-phase Lewis--Nielsen model unifies dielectric and thermal behavior
\item Pore shape factor $A=0.08$ fingerprints NIPS sponge vs.\ spherical pores
\item Single-parameter thermal fit: $\lamBN = \SI{6.8}{\watt\per\meter\per\kelvin}$, interface-limited
\item 70~wt\% h-BN gives the best measured \eps--\lam{} balance; model loci agree
\end{itemize}
